cmake_minimum_required(VERSION 3.22.1)
project(nanocode_jni CXX)
set(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../..")
add_library(nanocode_core STATIC IMPORTED)
set_target_properties(nanocode_core PROPERTIES IMPORTED_LOCATION "${CMAKE_CURRENT_SOURCE_DIR}/../rustLibs/${ANDROID_ABI}/libnanocode.a")
add_library(nanocode_jni SHARED nanocode_jni.cpp)
target_include_directories(nanocode_jni PRIVATE "${ROOT}/include")
target_link_libraries(nanocode_jni nanocode_core log dl m)
target_link_options(nanocode_jni PRIVATE "-Wl,-z,max-page-size=16384")
